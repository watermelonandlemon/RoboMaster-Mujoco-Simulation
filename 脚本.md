# 说明：这个脚本在Blender中运行，把.blender带颜色地导出.obj，供Mujoco使用。

import bpy
import shutil
import xml.etree.ElementTree as ET
from pathlib import Path
from collections import defaultdict

# ==========================================================
# 1. 路径设置
# ==========================================================

PROJECT_DIR = Path(
    r"D:\01-RoboMatser\workspace\blender\RMUC2027_MuJoCo_Starter"
)

OUTPUT_DIR = PROJECT_DIR / "meshes" / "colored"
XML_PATH = PROJECT_DIR / "scene_color.xml"

DART_SOURCE = (
    PROJECT_DIR / "meshes" / "optimized" / "dart_visual.stl"
)

# ==========================================================
# 2. 减面设置
# ==========================================================

USE_DECIMATE = True

DECIMATE_RATIO = 0.15
# 保留约15%的三角面

DECIMATE_MIN_POLYGONS = 300
# 面数少于300的物体不减面

# ==========================================================
# 3. 场景设置
# ==========================================================

UNIT_SCALE = bpy.context.scene.unit_settings.scale_length

DEFAULT_RGBA = (0.65, 0.65, 0.65, 1.0)

COLOR_DIGITS = 4

# 保留你目前的飞镖位置和方向
DART_POSITION = (1.0, 14.2, 0.75)
DART_EULER = (0.0, 90.0, 0.0)

# ==========================================================
# 4. 检查工程
# ==========================================================

if not bpy.data.filepath:
    raise RuntimeError("请先保存 Blender 工程！")

if not PROJECT_DIR.is_dir():
    raise FileNotFoundError(
        f"找不到 MuJoCo 工程目录：{PROJECT_DIR}"
    )

OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

# 只删除本脚本之前生成的 OBJ
for file in OUTPUT_DIR.glob("field_color_*.obj"):
    file.unlink()

print("\n========== 开始导出彩色场地 ==========")

print("Blender版本：", bpy.app.version_string)
print("网格单位缩放：", UNIT_SCALE)
print("减面比例：", DECIMATE_RATIO)

# ==========================================================
# 5. 检查是否有图片贴图
# ==========================================================

texture_materials = []

for mat in bpy.data.materials:
    if not mat.use_nodes or not mat.node_tree:
        continue

    if any(
        node.type == "TEX_IMAGE" and node.image
        for node in mat.node_tree.nodes
    ):
        texture_materials.append(mat.name)

if texture_materials:
    print("\n注意：发现图片贴图材质：")
    print(texture_materials)
    print("当前脚本按材质基础颜色导出，不烘焙图片贴图。")

# ==========================================================
# 6. 获取材质 RGB
# ==========================================================

def get_rgba(material):

    if material is None:
        return DEFAULT_RGBA

    color = material.diffuse_color

    return tuple(
        max(0.0, min(1.0, float(v)))
        for v in color
    )


# ==========================================================
# 7. 获取场景网格物体
# ==========================================================

source_objects = [
    obj for obj in bpy.context.scene.objects
    if obj.type == "MESH"
    and len(obj.data.polygons) > 0
    and not obj.hide_render
]

print("参与导出的网格物体：", len(source_objects))

if not source_objects:
    raise RuntimeError("没有找到可导出的网格物体！")

# ==========================================================
# 8. 材质颜色管理
# ==========================================================

color_to_id = {}
color_to_rgba = {}
color_to_label = {}

file_handles = {}
vertex_counts = defaultdict(int)

total_triangles = 0
total_vertices = 0

bbox_min = [float("inf")] * 3
bbox_max = [float("-inf")] * 3


def register_color(material):

    rgba = get_rgba(material)

    key = tuple(
        round(v, COLOR_DIGITS)
        for v in rgba
    )

    if key not in color_to_id:

        idx = len(color_to_id)

        color_to_id[key] = idx
        color_to_rgba[key] = rgba

        color_to_label[key] = (
            material.name if material else "NoMaterial"
        )

        filename = f"field_color_{idx:03d}.obj"

        path = OUTPUT_DIR / filename

        handle = path.open(
            "w",
            encoding="utf-8",
            newline="\n"
        )

        handle.write("# Generated from Blender for MuJoCo\n")

        file_handles[key] = handle

    return key


# ==========================================================
# 9. 创建临时集合
# 不修改原始 Blender 模型
# ==========================================================

temp_collection = bpy.data.collections.new(
    "__MUJOCO_EXPORT_TEMP__"
)

bpy.context.scene.collection.children.link(
    temp_collection
)

# ==========================================================
# 10. 逐个处理物体
# ==========================================================

try:

    for object_index, source_obj in enumerate(
        source_objects, start=1
    ):

        print(
            f"\n[{object_index}/{len(source_objects)}]"
            f" 处理：{source_obj.name}"
        )

        temp_obj = source_obj.copy()

        # 共享原始网格数据，不修改原模型
        temp_obj.data = source_obj.data

        temp_collection.objects.link(temp_obj)

        # 固定当前世界坐标变换
        temp_obj.parent = None

        temp_obj.matrix_world = (
            source_obj.matrix_world.copy()
        )

        # 清除临时对象上的约束和动画
        for constraint in list(temp_obj.constraints):
            temp_obj.constraints.remove(constraint)

        if temp_obj.animation_data:
            temp_obj.animation_data_clear()

        evaluated_obj = None
        evaluated_mesh = None

        try:

            # 添加临时减面修改器
            if (
                USE_DECIMATE
                and len(source_obj.data.polygons)
                >= DECIMATE_MIN_POLYGONS
            ):

                dec = temp_obj.modifiers.new(
                    name="MuJoCo_Decimate",
                    type="DECIMATE"
                )

                dec.decimate_type = "COLLAPSE"
                dec.ratio = DECIMATE_RATIO

            # 获取应用修改器后的模型
            bpy.context.view_layer.update()

            depsgraph = (
                bpy.context.evaluated_depsgraph_get()
            )

            evaluated_obj = temp_obj.evaluated_get(
                depsgraph
            )

            evaluated_mesh = evaluated_obj.to_mesh(
                preserve_all_data_layers=True,
                depsgraph=depsgraph
            )

            if evaluated_mesh is None:
                continue

            evaluated_mesh.calc_loop_triangles()

            world_matrix = (
                evaluated_obj.matrix_world.copy()
            )

            # 镜像变换需要调整三角面绕序
            mirrored = (
                world_matrix.to_3x3().determinant() < 0
            )

            # 按材质颜色分组
            triangles_by_color = defaultdict(list)

            for tri in evaluated_mesh.loop_triangles:

                vertices = tuple(
                    int(v) for v in tri.vertices
                )

                if len(set(vertices)) != 3:
                    continue

                if tri.area < 1e-18:
                    continue

                mat_index = int(tri.material_index)

                material = None

                if 0 <= mat_index < len(
                    evaluated_mesh.materials
                ):
                    material = evaluated_mesh.materials[
                        mat_index
                    ]

                elif 0 <= mat_index < len(
                    source_obj.data.materials
                ):
                    material = source_obj.data.materials[
                        mat_index
                    ]

                color_key = register_color(material)

                triangles_by_color[color_key].append(
                    vertices
                )

            # 分颜色写入 OBJ
            for color_key, triangles in (
                triangles_by_color.items()
            ):

                handle = file_handles[color_key]

                base_vertex = vertex_counts[color_key]

                local_vertex_map = {}

                for triangle in triangles:

                    face_indices = []

                    for vertex_index in triangle:

                        if vertex_index not in local_vertex_map:

                            local_co = (
                                evaluated_mesh.vertices[
                                    vertex_index
                                ].co
                            )

                            # 转换为 Blender 世界坐标
                            world_co = (
                                world_matrix @ local_co
                            )

                            # 转换为米
                            world_co = world_co * UNIT_SCALE

                            new_index = (
                                base_vertex
                                + len(local_vertex_map)
                                + 1
                            )

                            local_vertex_map[
                                vertex_index
                            ] = new_index

                            handle.write(
                                f"v {world_co.x:.9f} "
                                f"{world_co.y:.9f} "
                                f"{world_co.z:.9f}\n"
                            )

                            coords = (
                                world_co.x,
                                world_co.y,
                                world_co.z
                            )

                            for axis in range(3):

                                bbox_min[axis] = min(
                                    bbox_min[axis],
                                    coords[axis]
                                )

                                bbox_max[axis] = max(
                                    bbox_max[axis],
                                    coords[axis]
                                )

                        face_indices.append(
                            local_vertex_map[vertex_index]
                        )

                    if mirrored:
                        face_indices.reverse()

                    handle.write(
                        f"f {face_indices[0]} "
                        f"{face_indices[1]} "
                        f"{face_indices[2]}\n"
                    )

                vertex_counts[color_key] += (
                    len(local_vertex_map)
                )

                total_vertices += len(local_vertex_map)

                total_triangles += len(triangles)

        finally:

            if evaluated_obj is not None:
                evaluated_obj.to_mesh_clear()

            bpy.data.objects.remove(
                temp_obj,
                do_unlink=True
            )

finally:

    # 关闭 OBJ 文件
    for handle in file_handles.values():
        handle.close()

    # 清理所有临时对象
    for obj in list(temp_collection.objects):
        bpy.data.objects.remove(
            obj,
            do_unlink=True
        )

    bpy.data.collections.remove(
        temp_collection,
        do_unlink=True
    )

# ==========================================================
# 11. 生成 MuJoCo XML
# ==========================================================

if not color_to_id:
    raise RuntimeError("没有导出任何三角面！")

xml = []

xml.append(
    '<mujoco model="RMUC2027_colored">'
)

xml.append(
    '  <compiler angle="degree" '
    'meshdir="meshes/colored"/>'
)

xml.append(
    '  <statistic center="14 7.5 1" extent="22"/>'
)

xml.append(
    '  <option timestep="0.005" gravity="0 0 -9.81"/>'
)

# 资源定义
xml.append("  <asset>")

for color_key, idx in sorted(
    color_to_id.items(),
    key=lambda item: item[1]
):

    xml.append(
        f'    <mesh name="field_mesh_{idx:03d}" '
        f'file="field_color_{idx:03d}.obj" '
        f'inertia="shell"/>'
    )

# 复制飞镖模型到彩色模型目录
dart_output = OUTPUT_DIR / "dart_visual.stl"

has_dart = DART_SOURCE.is_file()

if has_dart:

    shutil.copy2(
        DART_SOURCE,
        dart_output
    )

    xml.append(
        '    <mesh name="dart_mesh" '
        'file="dart_visual.stl" '
        'scale="0.001 0.001 0.001"/>'
    )

else:

    print("\n警告：没有找到飞镖 STL：")
    print(DART_SOURCE)

xml.append("  </asset>")

# 仿真世界
xml.append("  <worldbody>")

xml.append(
    '    <light name="sun" type="directional" '
    'pos="14 7.5 18" dir="0 0 -1" '
    'diffuse="0.9 0.9 0.9" '
    'ambient="0.25 0.25 0.25" '
    'specular="0.1 0.1 0.1" '
    'castshadow="false"/>'
)

# 每个颜色组对应一个 MuJoCo 几何对象
for color_key, idx in sorted(
    color_to_id.items(),
    key=lambda item: item[1]
):

    rgba = color_to_rgba[color_key]

    rgba_text = " ".join(
        f"{v:.4f}" for v in rgba
    )

    xml.append(
        f'    <geom name="field_geom_{idx:03d}" '
        f'type="mesh" '
        f'mesh="field_mesh_{idx:03d}" '
        f'rgba="{rgba_text}" '
        f'contype="0" conaffinity="0"/>'
    )

# 临时地面
xml.append(
    '    <geom name="temporary_floor" '
    'type="plane" pos="14 7.5 -0.30" '
    'size="14 8 0.1" '
    'rgba="0.10 0.10 0.12 1"/>'
)

# 飞镖
if has_dart:

    pos_text = " ".join(
        f"{v:g}" for v in DART_POSITION
    )

    euler_text = " ".join(
        f"{v:g}" for v in DART_EULER
    )

    xml.append(
        f'    <body name="dart_preview" '
        f'pos="{pos_text}" euler="{euler_text}">'
    )

    xml.append(
        '      <geom name="dart_geom" '
        'type="mesh" mesh="dart_mesh" '
        'rgba="1 0.4 0.1 1" '
        'contype="0" conaffinity="0"/>'
    )

    xml.append("    </body>")

xml.append("  </worldbody>")
xml.append("</mujoco>")

# 写入 XML
XML_PATH.write_text(
    "\n".join(xml),
    encoding="utf-8"
)

# 检查 XML 格式
ET.parse(XML_PATH)

# ==========================================================
# 12. 输出报告
# ==========================================================

print("\n========== 导出成功 ==========")

print("颜色组数量：", len(color_to_id))
print("三角面数量：", total_triangles)
print("输出顶点数量：", total_vertices)

print("OBJ 输出目录：", OUTPUT_DIR)
print("XML 文件：", XML_PATH)

print("飞镖模型：", "已加入" if has_dart else "未加入")

if bbox_min[0] != float("inf"):

    dimensions = [
        bbox_max[i] - bbox_min[i]
        for i in range(3)
    ]

    print("模型最小坐标：", bbox_min)
    print("模型最大坐标：", bbox_max)
    print("场地尺寸 XYZ（米）：", dimensions)

print("\n各颜色组：")

for color_key, idx in sorted(
    color_to_id.items(),
    key=lambda item: item[1]
):

    print(
        f"field_color_{idx:03d}.obj",
        "RGB =", color_to_rgba[color_key],
        "材质 =", color_to_label[color_key]
    )

print("\nXML 格式检查：通过")
print("请回到 PyCharm 运行 scene_color.xml！")
